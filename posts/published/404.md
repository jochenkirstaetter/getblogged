    var path = window.location.pathname;
    var lowerPath = path.toLowerCase();
    if (path !== lowerPath) {
        var targetUrl = lowerPath + window.location.search + window.location.hash;
        var banner = document.getElementById('redirect-banner');
        var targetEl = document.getElementById('redirect-target');
        var countdownEl = document.getElementById('redirect-countdown');
        var redirectNowBtn = document.getElementById('redirect-now-btn');
        var cancelBtn = document.getElementById('redirect-cancel-btn');

        if (banner && targetEl && countdownEl) {
            targetEl.textContent = lowerPath;
            redirectNowBtn.href = targetUrl;
            banner.style.display = 'block';

            var secondsLeft = 4;
            var timer = setInterval(function() {
                secondsLeft--;
                if (countdownEl) countdownEl.textContent = secondsLeft;
                if (secondsLeft <= 0) {
                    clearInterval(timer);
                    window.location.replace(targetUrl);
                }
            }, 1000);

            if (redirectNowBtn) {
                redirectNowBtn.addEventListener('click', function(e) {
                    e.preventDefault();
                    clearInterval(timer);
                    window.location.replace(targetUrl);
                });
            }

            if (cancelBtn) {
                cancelBtn.addEventListener('click', function() {
                    clearInterval(timer);
                    banner.innerHTML = '<p style="margin: 0; color: #666;">Auto-redirect cancelled. You can continue browsing below or click <a href="' + targetUrl + '">here</a> when ready.</p>';
                });
            }
        }
    }
})();
</script>
